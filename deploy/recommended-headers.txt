# Recommended response headers for the static Ours frontend.
# A <meta> Content-Security-Policy is also injected into the production build.
# frame-ancestors and Strict-Transport-Security only take effect as real headers.

Content-Security-Policy: default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' data:; font-src 'self'; connect-src 'self' https://YOUR_PROJECT.supabase.co; base-uri 'self'; form-action 'self'; object-src 'none'; frame-ancestors 'none'
Referrer-Policy: no-referrer
X-Content-Type-Options: nosniff
X-Frame-Options: DENY
Permissions-Policy: camera=(), microphone=(), geolocation=(), payment=()
Strict-Transport-Security: max-age=63072000; includeSubDomains
Cache-Control: no-store
